# grab_url Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `asrgrab grab_url <url>` 一条命令完成「开浏览器 → 驱动点播视频 → 本地采集转写 → 收尾必通知」，产出与 `start` 相同的 transcript 与 `grab_end` 交接事件。

**Architecture:** 纯逻辑（参数/结束判定/事件模型）放 `GrabUrl.cs` 供单测；CDP 协议客户端 `Cdp.cs` 与 Edge 进程管理 `EdgeHost.cs` 薄封装；编排放 `Commands.GrabUrl`（沿用一命令一静态方法的既有模式），复用 `SessionRunner.Run` 跑采集，主线程等结束条件后写 `stop.flag` 收口。收尾（落稿→关浏览器→`grab_end`→`--notify-cmd`）放 `finally`，任何异常路径必达。

**Tech Stack:** C# / .NET 8（net8.0-windows, x64）、System.Net.WebSockets.Client、System.Text.Json、xunit（测试）。CDP 目标：Edge `--remote-debugging-port`（127.0.0.1:9222）。

**Spec:** `docs/superpowers/specs/2026-10-24-grab-url-design.md`（本计划任务编号与 spec §对应：CLI 契约 §2、流水线 §3、grab_end §4、notify §5、首登 §6、错误 §7、限制 §8、落点 §9、测试 §10、同步 §11）。

```csharp
namespace AsrGrab;

public enum EndReason { Ended, ClipSwitch, MaxSeconds, Error }

public sealed class GrabUrlOptions
{
    public string Url = "";
    public int MaxSeconds;
    public int ChunkSec = 12;
    public string Model = "small";
    public bool NoDiarize;
    public int NumSpeakers;
    public int VideoTimeoutSec = 30;
    public string? NotifyCmd;
    public bool KeepBrowser;
    public int Port = 9222;

    public static GrabUrlOptions Parse(string[] args)
    {
        var ap = new ArgParser(args);
        var o = new GrabUrlOptions();
        if (ap.Positional.Count < 1 || string.IsNullOrWhiteSpace(ap.Positional[0]))
            throw new CliError(ExitCodes.Usage, "USAGE", "grab_url 需要 <url>", new()
            {
                ["hint"] = "asrgrab grab_url <url> [--max-seconds N] [--notify-cmd \"<cmd>\"]",
            });
        o.Url = ap.Positional[0];
        if (!o.Url.StartsWith("http://", StringComparison.OrdinalIgnoreCase) &&
            !o.Url.StartsWith("https://", StringComparison.OrdinalIgnoreCase))
            throw new CliError(ExitCodes.Usage, "USAGE", "url 必须以 http(s):// 开头", new() { ["url"] = o.Url });
        o.MaxSeconds = ap.Int("max-seconds", 0);
        o.ChunkSec = Math.Max(1, ap.Int("chunk", 12));
        o.Model = ap.Str("model", "small");
        o.NoDiarize = ap.Flag("no-diarize");
        o.NumSpeakers = ap.Int("num-speakers", 0);
        o.VideoTimeoutSec = Math.Max(3, ap.Int("video-timeout", 30));
        o.NotifyCmd = ap.Get("notify-cmd");
        o.KeepBrowser = ap.Flag("keep-browser");
        o.Port = ap.Int("port", 9222);
        return o;
    }
}
```

```csharp
public static class GrabEnd
{
    public static Dictionary<string, object?> Build(
        EndReason reason, int exitCode, string sessionId, string? outFile,
        double elapsedSec, int segments, string? error)
        => new()
        {
            ["event"] = "grab_end",
            ["end_reason"] = reason switch
            {
                EndReason.Ended => "ended",
                EndReason.ClipSwitch => "clip-switch",
                EndReason.MaxSeconds => "max-seconds",
                _ => "error",
            },
            ["exit_code"] = exitCode,
            ["session_id"] = sessionId,
            ["out_file"] = outFile,
            ["elapsed_s"] = Math.Round(elapsedSec, 1),
            ["segments"] = segments,
            ["error"] = error,
        };
}
```

```csharp
public static class EndCondition
{
    public sealed record VideoSample(double CurrentTimeSec, double DurationSec, bool Ended, string? Src);

    /// 返回 (是否停止, 原因)。优先级：ended > clip-switch > max-seconds（spec §3）。
    public static (bool Stop, EndReason? Reason) Evaluate(
        VideoSample s, string? firstSrc, double elapsedSec, int maxSeconds)
    {
        if (s.Ended) return (true, EndReason.Ended);
        if (firstSrc != null && s.Src != null && s.Src != firstSrc) return (true, EndReason.ClipSwitch);
        if (maxSeconds > 0 && elapsedSec >= maxSeconds) return (true, EndReason.MaxSeconds);
        return (false, null);
    }
}
```

```csharp
using System.Diagnostics;
using System.Net.Sockets;

namespace AsrGrab;

/// Edge 进程生命周期：端口探测 / 拉起（launch_edge.bat 同款参数）/ 关闭。
public sealed class EdgeHost : IDisposable
{
    private readonly Process _proc;
    private EdgeHost(Process proc) => _proc = proc;

    public static bool PortOpen(int port)
    {
        try
        {
            using var c = new TcpClient();
            c.Connect("127.0.0.1", port);
            return true;
        }
        catch { return false; }
    }

    public static EdgeHost Launch(int port)
    {
        var profile = Path.Combine(
            Environment.GetFolderPath(Environment.SpecialFolder.UserProfile),
            ".asrgrab", "edge-profile");
        foreach (var exe in new[]
        {
            @"C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe",
            @"C:\Program Files\Microsoft\Edge\Application\msedge.exe",
        })
        {
            if (!File.Exists(exe)) continue;
            var psi = new ProcessStartInfo { FileName = exe, UseShellExecute = false };
            psi.ArgumentList.Add($"--remote-debugging-port={port}");
            psi.ArgumentList.Add("--remote-allow-origins=*");
            psi.ArgumentList.Add($"--user-data-dir={profile}");
            psi.ArgumentList.Add("--no-first-run");
            psi.ArgumentList.Add("--no-default-browser-check");
            psi.ArgumentList.Add("--proxy-bypass-list=127.0.0.1;localhost");
            psi.ArgumentList.Add("--disable-features=msEdgeStartupBoost");
            psi.ArgumentList.Add("about:blank");
            var p = Process.Start(psi)
                ?? throw new CliError(ExitCodes.Dependency, "DEPENDENCY", "无法启动 Edge");
            return new EdgeHost(p);
        }
        throw new CliError(ExitCodes.Dependency, "DEPENDENCY", "未找到 msedge.exe");
    }

    public void Close()
    {
        try
        {
            _proc.CloseMainWindow();
            if (!_proc.WaitForExit(3000)) _proc.Kill(entireProcessTree: true);
        }
        catch { /* 进程已退出 */ }
    }

    public void Dispose() => Close();
}
```

```python
import math, struct, wave

def make(path, sec, freq):
    with wave.open(path, "w") as w:
        w.setnchannels(1); w.setsampwidth(2); w.setframerate(16000)
        frames = b"".join(
            struct.pack("<h", int(8000 * math.sin(2 * math.pi * freq * i / 16000)))
            for i in range(int(16000 * sec)))
        w.writeframes(frames)

make("a.wav", 6, 440)
make("b.wav", 6, 660)
print("ok")
```

```csharp
          const v = vs.find(x => x.duration > 0) || vs[0];
          if (!v) return { no_video: true };
          return {
            no_video: false,
            paused: v.paused,
            ended: v.ended,
            ct: v.currentTime,
            dur: isFinite(v.duration) ? v.duration : 0,
            live: !isFinite(v.duration),
            muted: v.muted,
            src: (v.currentSrc || '').slice(0, 200),
          };
        })())
        """;

    private const string PlayJs = """
        JSON.stringify((() => {
          const vs = Array.from(document.querySelectorAll('video'));
          const v = vs.find(x => x.duration > 0) || vs[0];
          if (!v) return { no_video: true };
          v.muted = false;
          const p = v.play();
          return { called: true, err: p && p.catch ? 'async' : 'sync' };
        })())
        """;

    private const string PauseSeek0Js = """
        JSON.stringify((() => {
          const vs = Array.from(document.querySelectorAll('video'));
          const v = vs.find(x => x.duration > 0) || vs[0];
          if (!v) return { no_video: true };
          v.pause();
          v.currentTime = 0;
          v.muted = false;
          return { called: true };
        })())
        """;

    private readonly ClientWebSocket _ws = new();
    private int _id;

    private CdpClient() { }

    public static CdpClient Attach(int port, int waitSec = 15)
    {
        var deadline = DateTime.UtcNow.AddSeconds(waitSec);
        Exception? last = null;
        while (DateTime.UtcNow < deadline)
        {
            try
            {
                using var http = new HttpClient { Timeout = TimeSpan.FromSeconds(2) };
                var list = http.GetStringAsync($"http://127.0.0.1:{port}/json/list").GetAwaiter().GetResult();
                using var doc = JsonDocument.Parse(list);
                string? wsUrl = null;
                foreach (var t in doc.RootElement.EnumerateArray())
                {
                    if (t.GetProperty("type").GetString() != "page") continue;
                    wsUrl = t.GetProperty("webSocketDebuggerUrl").GetString();
                    break;
                }
                if (wsUrl == null) { Thread.Sleep(400); continue; }
                var c = new CdpClient();
                c._ws.ConnectAsync(new Uri(wsUrl), CancellationToken.None).GetAwaiter().GetResult();
                return c;
            }
            catch (Exception e) { last = e; Thread.Sleep(400); }
        }
        throw new CliError(ExitCodes.Dependency, "DEPENDENCY", $"无法连接 CDP :{port}",
            new() { ["detail"] = last?.Message });
    }

    private JsonElement Call(string method, object? parms)
    {
        var id = Interlocked.Increment(ref _id);
        var payload = JsonSerializer.Serialize(new { id, method, @params = parms });
        var bytes = Encoding.UTF8.GetBytes(payload);
        _ws.SendAsync(bytes, WebSocketMessageType.Text, true, CancellationToken.None).GetAwaiter().GetResult();
        var buf = new byte[1 << 20];
        while (true)
        {
            using var ms = new MemoryStream();
            WebSocketReceiveResult r;
            do
            {
                r = _ws.ReceiveAsync(buf, CancellationToken.None).GetAwaiter().GetResult();
                ms.Write(buf, 0, r.Count);
            } while (!r.EndOfMessage);
            using var doc = JsonDocument.Parse(ms.ToArray());
            if (!doc.RootElement.TryGetProperty("id", out var got) || got.GetInt32() != id) continue;
            if (doc.RootElement.TryGetProperty("error", out var err))
                throw new CliError(ExitCodes.Engine, "CDP_ERROR", err.ToString());
            return doc.RootElement.GetProperty("result").Clone();
        }
    }

    public string EvalJs(string expression)
    {
        var result = Call("Runtime.evaluate", new
        {
            expression,
            returnByValue = true,
            awaitPromise = true,
        });
        if (!result.TryGetProperty("result", out var inner)) return "";
        return inner.TryGetProperty("value", out var v) && v.ValueKind == JsonValueKind.String
            ? v.GetString() ?? "" : "";
    }

    public void Navigate(string url) => Call("Page.navigate", new { url });

    public VideoState Probe()
    {
        var raw = EvalJs(ProbeJs);
        if (string.IsNullOrEmpty(raw)) return new VideoState { NoVideo = true };
        using var d = JsonDocument.Parse(raw);
        var e = d.RootElement;
        if (e.TryGetProperty("no_video", out var nv) && nv.GetBoolean())
            return new VideoState { NoVideo = true };
        return new VideoState
        {
            Paused = e.GetProperty("paused").GetBoolean(),
            Ended = e.GetProperty("ended").GetBoolean(),
            CurrentTime = e.GetProperty("ct").GetDouble(),
            Duration = e.GetProperty("dur").GetDouble(),
            Muted = e.GetProperty("muted").GetBoolean(),
            Src = e.GetProperty("src").GetString() ?? "",
            IsLive = e.GetProperty("live").GetBoolean(),
        };
    }

    public void Play() => EvalJs(PlayJs);

    public void PauseSeek0Unmute() => EvalJs(PauseSeek0Js);

    public void Dispose()
    {
        try { _ws.Abort(); _ws.Dispose(); } catch { }
    }
}
```

```csharp
    /// 等待顶层 video 就绪（duration>0）。直播（duration 非有限）直接返回由调用方拒绝。
    /// 超时抛 CliError(SOURCE)，附首登提示（spec §6）。
    public VideoState WaitForVideo(int timeoutSec)
    {
        var deadline = DateTime.UtcNow.AddSeconds(timeoutSec);
        while (DateTime.UtcNow < deadline)
        {
            VideoState s;
            try
            {
                s = Probe();
            }
            catch (CliError e) when (e.Code is "EVAL_ERROR" or "CDP_ERROR" or "CDP_TIMEOUT")
            {
                // 页面跳转/上下文重建等瞬态错误：继续轮询到截止
                Thread.Sleep(500);
                continue;
            }
            if (s.IsLive) return s;               // 调用方判 live → 拒绝
            if (!s.NoVideo && s.Duration > 0) return s;
            Thread.Sleep(500);
        }
        throw new CliError(ExitCodes.Source, "SOURCE", $"等待视频超时（{timeoutSec}s）", new()
        {
            ["hint"] = "可能需要登录：asrgrab grab_url <url> --keep-browser 打开页面手动登录后重跑；或加大 --video-timeout",
        });
    }

    /// 轮询结束条件（1s 一次）；返回触发的 EndReason。CDP 中断按 error 返回。
    public EndReason WaitForEnd(string? firstSrc, int maxSeconds, Stopwatch sw)
    {
        while (true)
        {
            EndCondition.VideoSample sample;
            try
            {
                var s = Probe();
                sample = new EndCondition.VideoSample(s.CurrentTime, s.Duration, s.Ended, s.Src);
            }
            catch (Exception)
            {
                return EndReason.Error;
            }
            var (stop, reason) = EndCondition.Evaluate(sample, firstSrc, sw.Elapsed.TotalSeconds, maxSeconds);
            if (stop && reason != null) return reason.Value;
            Thread.Sleep(1000);
        }
    }
```

```csharp
    public static int GrabUrl(string[] args)
    {
        var sw = System.Diagnostics.Stopwatch.StartNew();
        var opt = GrabUrlOptions.Parse(args);
        Out.H($"grab_url: {opt.Url}");

        var cur = SessionStore.ReadCurrent();
        if (cur != null && SessionStore.PidAlive(cur.Pid))
            throw new CliError(ExitCodes.Busy, "BUSY", "已有会话在运行", new() { ["session_id"] = cur.SessionId });

        var sources = AudioSourceEnumerator.List();
        var source = SourceResolver.Default(sources)
            ?? throw new CliError(ExitCodes.Source, "SOURCE", "未找到任何可用音源（检查 Windows 声音设备）");

        var sessionId = SessionStore.NowId();
        var outDir = "transcripts";
        var mdPath = Path.Combine(outDir, $"transcript_{sessionId}.md");

        EdgeHost? owned = null;
        CdpClient? cdp = null;
        Task? sessionTask = null;
        var reason = EndReason.Error;
        string? error = null;
        var exit = ExitCodes.Ok;
        var segments = 0;

        try
        {
            // 1. 浏览器：附着或自开（spec §3.1）
            if (EdgeHost.PortOpen(opt.Port))
            {
                Out.H($"附着已有调试实例 :{opt.Port}（收尾不关）");
            }
            else
            {
                Out.H("启动 Edge…");
                owned = EdgeHost.Launch(opt.Port);
            }
            cdp = CdpClient.Attach(opt.Port);

            // 2. 打开并等 video（spec §3.2）
            cdp.Navigate(opt.Url);
            var vs = cdp.WaitForVideo(opt.VideoTimeoutSec);
            if (vs.IsLive)
                throw new CliError(ExitCodes.Source, "LIVE_UNSUPPORTED",
                    "直播流 v1 不支持（duration=Infinity）；grab_url 仅支持点播页");
            Out.H($"视频就绪 dur={vs.Duration:0.#}s src={vs.Src}");

            // 3. 对齐：pause → seek(0) → unmute（不动 volume，spec §3.3）
            cdp.PauseSeek0Unmute();

            // 4. 采集会话跑在后台线程
            var cfg = new SessionConfig
            {
                SessionId = sessionId,
                Source = source,
                OutDir = outDir,
                ChunkSec = opt.ChunkSec,
                Model = NormalizeModel(opt.Model),
                NoDiarize = opt.NoDiarize,
                NumSpeakers = opt.NumSpeakers,
            };
            sessionTask = Task.Run(() => SessionRunner.Run(cfg, ev => Out.Event(ev)));
            Out.H("采集已启动，开播…");

            // 3.5 先录后播（spec §3.3）：采集已启动再 play；被自动播放策略拒绝仅告警
            if (!cdp.PlayAndVerify())
                Out.H("警告：play() 后进度未推进（可能被浏览器自动播放策略拦截）；继续采集");

            // 5. 等结束（ended / clip-switch / max-seconds，spec §3.4）
            reason = cdp.WaitForEnd(vs.Src, opt.MaxSeconds, sw);
            Out.H($"结束：{reason}");
            exit = ExitCodes.Ok;
        }
        catch (CliError e)
        {
            reason = EndReason.Error;
            error = e.Message;
            exit = e.ExitCode;
            Out.Error(e);
        }
        catch (Exception e)
        {
            reason = EndReason.Error;
            error = e.Message;
            exit = ExitCodes.Internal;
            Out.Error(new CliError(ExitCodes.Internal, "INTERNAL", e.Message));
        }
        finally
        {
            // 收尾必达（spec §3.5 / §7）：stop 会话 → 关自开浏览器 → grab_end → notify
            if (sessionTask != null)
            {
                try
                {
                    var flag = Path.Combine(SessionStore.SessionDir(sessionId), "stop.flag");
                    File.WriteAllText(flag, "stop");
                    if (!sessionTask.Wait(TimeSpan.FromSeconds(60)))
                        Out.H("警告：会话收口超时");
                    segments = SessionStore.GetInt(
                        JsonIo.Read<Dictionary<string, object?>>(
                            Path.Combine(SessionStore.SessionDir(sessionId), "status.json")) ?? new(),
                        "segments");
                }
                catch (Exception e) { Out.H($"收口异常：{e.Message}"); }
            }
            if (owned != null && !opt.KeepBrowser) owned.Close();
            else if (opt.KeepBrowser) Out.H("已按 --keep-browser 保留浏览器");
            cdp?.Dispose();

            var outFile = sessionTask != null ? mdPath : null;
            Out.Event(GrabEnd.Build(reason, exit, sessionId, outFile, sw.Elapsed.TotalSeconds, segments, error));
            if (opt.NotifyCmd != null) RunNotify(opt.NotifyCmd, exit, sessionId, outFile);
        }
        return exit;
    }

    private static void RunNotify(string cmd, int exitCode, string sessionId, string? outFile)
    {
        try
        {
            var psi = new ProcessStartInfo
            {
                FileName = "cmd.exe",
                UseShellExecute = false,
            };
            psi.ArgumentList.Add("/c");
            psi.ArgumentList.Add(cmd);
            psi.Environment["ASRGRAB_EXIT"] = exitCode.ToString();
            psi.Environment["ASRGRAB_SESSION_ID"] = sessionId;
            psi.Environment["ASRGRAB_OUT_MD"] = outFile ?? "";
            psi.Environment["ASRGRAB_OUT_JSON"] = outFile == null ? "" : Path.ChangeExtension(outFile, ".json");
            using var p = Process.Start(psi);
            if (p == null || !p.WaitForExit(30_000))
                Console.Error.WriteLine("警告：--notify-cmd 未在 30s 内完成（不影响主退出码）");
        }
        catch (Exception e)
        {
            Console.Error.WriteLine($"警告：--notify-cmd 执行失败：{e.Message}（不影响主退出码）");
        }
    }
```
